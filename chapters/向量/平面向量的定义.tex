\section{向量}

\begin{definition}[向量]
    既有大小又有方向的量称为向量。
\end{definition}

不同于之前研究的量，向量之间的关系更复杂：
\begin{enumerate}
    \item 相等：大小和方向都相等
    \item 平行或者说是共线：方向相等或者相反
\end{enumerate}

\begin{definition}[零向量与单位向量]
    特别地我们定义零向量：$\overrightarrow{0}$是大小为$0$，方向任意的向量。此外我们定义长度为 $1$ 的向量为单位向量。
\end{definition}

\begin{theorem}[运算律]
    不难验证，这样定义的向量运算满足：
    \begin{multicols}{2}
        \begin{enumerate}
            \item 加法交换律：$\overrightarrow{a}+\overrightarrow{b} = \overrightarrow{b}+\overrightarrow{a}$
            \item 加法结合律：$\overrightarrow{a}+(\overrightarrow{b}+\overrightarrow{c}) = (\overrightarrow{a}+\overrightarrow{b})+\overrightarrow{c}$
            \item 加法有零元：$\overrightarrow{a}+\overrightarrow{0} = \overrightarrow{a}$
                  \columnbreak
            \item 数乘向量分配律：$\lambda(\overrightarrow{a}+\overrightarrow{b}) = \lambda\overrightarrow{a}+\lambda\overrightarrow{b}$
            \item 数乘标量分配律：$(\lambda_1+\lambda_2)\overrightarrow{a} = \lambda_1\overrightarrow{a}+\lambda_2\overrightarrow{a}$
            \item 数乘结合律：$\lambda_1 (\lambda_2\overrightarrow{a}) = (\lambda_1\lambda_2)\overrightarrow{a}$
            \item 数乘有单位元：$1 \overrightarrow{a} = \overrightarrow{a}$
        \end{enumerate}
    \end{multicols}
\end{theorem}

\begin{example}
    已知 $\overrightarrow{a}$、$\overrightarrow{b}$ 是不平行的向量，若 $\overrightarrow{A B}=\overrightarrow{a}+2 \overrightarrow{b}, \overrightarrow{B C}=-4 \overrightarrow{a}-\overrightarrow{b}, \overrightarrow{C D}=-5 \overrightarrow{a}-3 \overrightarrow{b}$ ，则下列关系中正确的是 （\qquad）
    \begin{tasks}(4)
        \task $\overrightarrow{A D}=\overrightarrow{C B}$
        \task $\overrightarrow{A D}=\overrightarrow{B C}$
        \task $\overrightarrow{A D}=2 \overrightarrow{B C}$
        \task $\overrightarrow{A D}=-2 \overrightarrow{B C}$
    \end{tasks}
\end{example}

    $\overrightarrow{a}-\overrightarrow{b} = \overrightarrow{a}+(-\overrightarrow{b})$

\end{definition}

\subsection{内积}
内积也叫数量积。
\begin{definition}[内积]
    $$
        \overrightarrow{a}\cdot \overrightarrow{b} = |\overrightarrow{a}||\overrightarrow{b}|\cos \langle \overrightarrow{a},\overrightarrow{b} \rangle
    $$
    其中$\langle \overrightarrow{a},\overrightarrow{b} \rangle \in [0,\pi]$是两向量的夹角，$|\overrightarrow{a}|$是向量的模长。此外，规定任何向量和零向量的内积都是0。
\end{definition}

\begin{theorem}[内积的运算律]
    不难验证内积满足下列运算律：
    \begin{enumerate}
        \item 交换律：$\overrightarrow{a}\cdot \overrightarrow{b} = \overrightarrow{b} \cdot \overrightarrow{a}$
        \item 分配律：$\overrightarrow{a}\cdot (\overrightarrow{b}+\overrightarrow{c}) = \overrightarrow{a} \cdot \overrightarrow{b} + \overrightarrow{a} \cdot \overrightarrow{c}$
        \item 数乘结合律：$(\lambda \overrightarrow{a})\cdot \overrightarrow{b} = \lambda (\overrightarrow{a} \cdot \overrightarrow{b})$
    \end{enumerate}
\end{theorem}

\begin{example}
    已知 $|\overrightarrow{a}|=|\overrightarrow{b}|=1$ ，且 $\overrightarrow{a}$ 与 $\overrightarrow{b}$ 的夹角为 $\frac{\pi}{3}$ ，则 $\overrightarrow{b}$ 与 $\overrightarrow{a}+\overrightarrow{b}$ 的夹角为 （\qquad）
    \begin{tasks}(4)
        \task $\frac{\pi}{12}$
        \task $\frac{\pi}{6}$
        \task $\frac{7 \pi}{12}$
        \task $\frac{5 \pi}{6}$
    \end{tasks}
\end{example}

\section{向量的坐标表示}
直观地说，我们可以把向量放在平面直角坐标系坐标系中，在有大小方向的量和二元坐标之间建立一一对应的关系。

数学上，我们有向量基本定理：
\begin{theorem}
    如果$\overrightarrow{e_1}, \overrightarrow{e_2}$是平面内两个不共线的向量，那么该平面内任何一个向量都可以被他们俩\textbf{唯一地}线性表示：
    $$
        \overrightarrow{a} = x\overrightarrow{e_1} + y \overrightarrow{e_2}
    $$
    我们称$\overrightarrow{e_1}, \overrightarrow{e_2}$是平面内的一组基。称二元组$(x,y)$是$\overrightarrow{a}$在这组基下的坐标。

    特别地，如果它们的都是单位向量，并且夹角是$\pi / 2$，称为单位正交基。
\end{theorem}

\begin{example}
    已知向量 $\overrightarrow{a}, \overrightarrow{b}$ 满足 $|\overrightarrow{a}|=1, \overrightarrow{b}=(-1, \sqrt{2})$ ，且 $\overrightarrow{a}, \overrightarrow{b}$ 的夹角为 $\frac{\pi}{6}$ ，则 $|2 \overrightarrow{a}-\overrightarrow{b}|=$ （\qquad）
    \begin{tasks}(4)
        \task $\sqrt{2}$
        \task 1
        \task 2
        \task 4
    \end{tasks}
\end{example}

\v{6em}

\begin{theorem}[Cauchy不等式]
    $$
        -|\overrightarrow{a}||\overrightarrow{b}| \le \overrightarrow{a} \cdot \overrightarrow{b} \le |\overrightarrow{a}||\overrightarrow{b}|
    $$
    使用坐标表示则是：
    $$
        -\sqrt{x_1^2+y_1^2}\sqrt{x_2^2+y_2^2} \le x_1x_2 + y_1y_2 \le \sqrt{x_1^2+y_1^2}\sqrt{x_2^2+y_2^2}
    $$
    更一般的，实际上这个结论对$n$维向量都成立，这也就是我们之前证明的Cauchy不等式。
\end{theorem}

\begin{theorem}[定比分点]

\subsection{三角形五心的向量表示}

以下设$a,b,c$是三角形ABC的三边长。O是三角形所在平面的一点。

\begin{theorem}
    O是外心$\iff |\overrightarrow{OA}| = |\overrightarrow{OB}| = |\overrightarrow{OC}|$
\end{theorem}

\begin{proof}
    这是显然的，OA、OB、OC就是外接圆半径，长度自然相等。
\end{proof}

\begin{theorem}
    O是重心$\iff \overrightarrow{OA} + \overrightarrow{OB} + \overrightarrow{OC} = \overrightarrow{0}$
\end{theorem}

\begin{proof}
    设M是AB中点，根据定比分点，$\overrightarrow{OA} + \overrightarrow{OB} = 2 \overrightarrow{OM}$，结合我们的条件也就是：
    $$
        2 \overrightarrow{OM} = -\overrightarrow{OC}
    $$
    也就是说O是CM的靠近M的三等分点，这显然就是重心。
\end{proof}

\begin{theorem}
    O是垂心$\iff \overrightarrow{OA} \cdot \overrightarrow{OB} = \overrightarrow{OA} \cdot \overrightarrow{OC} = \overrightarrow{OC} \cdot \overrightarrow{OB}$
\end{theorem}

\begin{proof}
    这是显然的：
    $$
        \overrightarrow{OA} \cdot \overrightarrow{OB} = \overrightarrow{OA} \cdot \overrightarrow{OC} \iff \overrightarrow{OA} \cdot \overrightarrow{CB} = 0 \iff OA \perp CB
    $$
\end{proof}

\begin{theorem}
    O是内心$\iff a\overrightarrow{OA} + b\overrightarrow{OB} + c\overrightarrow{OC} = \overrightarrow{0}$
\end{theorem}

\begin{proof}
    这是比较容易的：
    $$
        \begin{aligned}
                     & a\overrightarrow{OA} + b\overrightarrow{OB} + c\overrightarrow{OC} = \overrightarrow{0}               \\
            \implies & (a+b+c)\overrightarrow{OA} +b\overrightarrow{AB}+c\overrightarrow{AB} = \overrightarrow{0}            \\
            \implies & \overrightarrow{AO} = \frac{1}{a+b+c}(b\overrightarrow{AB}+c\overrightarrow{AC})                      \\
            \implies & \overrightarrow{AO} = \frac{bc}{a+b+c}(\frac{1}{c}\overrightarrow{AB}+\frac{1}{b}\overrightarrow{AC})
        \end{aligned}
    $$
    其中
    $$
        \frac{1}{c}\overrightarrow{AB}, \quad \frac{1}{b}\overrightarrow{AC}
    $$
    分别是AB、AC方向的单位向量，根据平行四边形定则（这里由于都是单位向量，因此是一个菱形），它们相加得到的AO就一定在角平分线上了。

    $\implies$：这就不那么显然了。

    我们首先介绍两个引理
    \begin{lemma}[角平分线]
        在三角形ABC中，D在BC上，设AD是角A的平分线，那么：
        $$
            \frac{DB}{DC} = \frac{c}{b}
        $$
        此引理使用正弦定理很容易证明。
    \end{lemma}
    \begin{lemma}[内心]
        在三角形ABC中，D在BC上，设AD是角A的平分线，O是内心那么：
        $$
            \frac{OD}{OA} = \frac{a}{b+c}
        $$
        此引理使用正弦定理很容易证明。
    \end{lemma}
    那么我们就有
    $$
        \begin{aligned}
                     & a\overrightarrow{OA} + b\overrightarrow{OB} + c\overrightarrow{OC}                                          \\
            \implies & a\overrightarrow{OA} +b(\overrightarrow{OD}+\overrightarrow{DB})+c(\overrightarrow{OD}+\overrightarrow{DC}) \\
            \implies & a\overrightarrow{OA} + (b+c)\overrightarrow{OD} + b\overrightarrow{DB}+c\overrightarrow{DC}
        \end{aligned}
    $$
    根据第一个引理：
    $$
