\section{点集}

\begin{definition}[点集]
    由一个或多个点组成的集合称为点集。点集可以是有限的，也可以是无限的。
\end{definition}

\begin{example}
    函数 $y=2 x^2-x+1$ 的图像上所有的点组成的集合．
\end{example}

\v{4em}

\begin{example}
    用适当的方法翻译这些集合，让它们更加直观。
    \begin{itemize}
        \item （1）$A=\left\{(x, y) \mid x+y=4, x \in \mathrm{~N}^*, y \in \mathrm{~N}^*\right\}$ ；
        \item （2）$B=\left\{\left.\frac{6}{1+x} \in \mathrm{Z} \right\rvert\, x \in \mathrm{~N}\right\}$ ；
        \item （3）平面直角坐标系中所有第二象限内的点．
    \end{itemize}
\end{example}
