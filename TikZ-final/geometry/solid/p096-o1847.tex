\begin{tikzpicture}[scale=1, every node/.style={font=\small}, line cap=round, line join=round]
  % regular tetrahedron-like pyramid S-ABC, SO perp ABC, O centre of equilateral ABC
  \coordinate (A) at (0,0);
  \coordinate (C) at (4.97,0);
  \coordinate (B) at (3.27,-1.82);
  \coordinate (O) at ($1/3*(A)+1/3*(B)+1/3*(C)$);
  \coordinate (S) at ($(O)+(0,4.0)$);

  \draw (S)--(A)--(B)--(C)--cycle;
  \draw (S)--(B);
  \draw[dashed] (A)--(C) (S)--(O) (A)--(O);

  \node[above] at (S) {$S$};
  \node[left] at (A) {$A$};
  \node[below] at (B) {$B$};
  \node[right] at (C) {$C$};
  \node[below] at (O) {$O$};
